## Global Constraints

Bu bölüm her task'ın gereksinimlerine örtük olarak dahildir. Plan 1'deki tüm kurallar aynen geçerlidir; burada yalnızca yeni/özelleşenler listelenir.

- **Plan 1 kurallarının tümü miras alınır:** `Result<T>` dönüş, `can()` veriye dokunmadan önce, Zod giriş doğrulama, soft delete (içerik varlıklarında `archivedAt`, join tablolarında gerçek silme), para alanları `@db.Decimal(12,2)`, dört ekran durumu, `'use server'` modül disiplini, "Sadece ben" etiketi.
- **Yeni `'use server'` modülleri yalnızca action export eder.** Sorgu yardımcıları `*-query.ts` dosyalarında, saf fonksiyonlar `src/lib/` altında durur.
- **Hocuspocus bağlantısı bir yetki sınırıdır.** `onAuthenticate` içinde oturum doğrulanır ve `can()` çalıştırılır; reddedilen bağlantı kapatılır. Arayüz "sadece yazabilenler websocket açar" kuralını görmezden gelemez — sunucu yine de kontrol eder.
- **Doküman görünürlük etiketi arayüzde:** `PUBLIC` → "Tüm kulüp", `CHANNEL` → "Kanal üyeleri", `PRIVATE` → "Sadece ben" (+ alt not "Yöneticiler erişebilir"). Kod içinde enum `PRIVATE` kalır.
- **`VIEW` paylaşımı websocket açmaz.** Salt-okur paylaşılan doküman, sunucudan getirilen ydoc ile `editable={false}` editörde gösterilir; canlı imleç ve düzenleme yalnızca yazma yetkisi olanlara.
- **`Activity.channelId` artık doldurulur** (içerik varlıkları için). Ana sayfa akışı, PRIVATE kanal hareketlerini üye olmayana göstermez.
- **Yeni varlık action'ları mevcut `content:*` yetkilerini yeniden kullanır** (`task`, `event`, `sponsor` için). Yalnızca gerçekten farklı kurallar yeni `Action` ekler: `document:*` (Plan 2) ve `budget:*` (Plan 4).
- **Full-text arama `tsvector` generated column + GIN index ile.** Prisma generated tsvector'i tanımaz: şema yalnızca düz metin kolonunu (`searchText`, `notes`, `description`) tutar; tsvector kolonu ve index'i migration dosyasına elle eklenen raw SQL ile gelir.
- **Sıralama (rank) Lexorank string.** `rank` `String` kolonu, sözlük sırasına göre sıralanır. `rankBetween(a, b)` sınır durumları ve yeniden dengeleme (rebalance) birim testle kapatılır. Sayısal fractional index (0.5, 0.75) precision'ı ~30 eklemede tüketir; string tabanlıdır ve deterministiktir.
- **Dosya yüklemeleri** `UPLOAD_DIR` (varsayılan `./uploads`) altına yazılır; `Attachment` satırı yalnızca `can()` doğrulamasından sonra dönen ID ile servis edilir. Yükleme ve indirme route handler'ları `getActor()` + `can()` çağırır.

```ts
// src/lib/doc-search-text.ts
import * as Y from 'yjs'

/**
 * BlockNote'un ydoc'u: kökte 'document' XmlFragment'i, içinde bloklar
 * (XmlElement), her blokta 'text' adlı XmlElement — onun da çocukları
 * metin düğümleri (XmlText). Sıralı gezintiyle blok metinlerini
 * yeni satırla birleştirir; arama metni bundan üretilir.
 */
export function extractTextFromYDoc(doc: Y.Doc): string {
  const root = doc.getXmlFragment('document')
  const parts: string[] = []
  for (let i = 0; i < root.length; i++) {
    const child = root.get(i)
    if (child instanceof Y.XmlElement) {
      parts.push(collectText(child))
    } else if (child instanceof Y.XmlText) {
      parts.push(child.toString())
    }
  }
  return parts.join('\n').trim()
}

function collectText(el: Y.XmlElement): string {
  const parts: string[] = []
  for (let i = 0; i < el.length; i++) {
    const child = el.get(i)
    if (child instanceof Y.XmlText) parts.push(child.toString())
    else if (child instanceof Y.XmlElement) parts.push(collectText(child))
  }
  return parts.join(' ')
}
```

```ts
import { Database } from '@hocuspocus/extension-database'
import { PrismaClient } from '../generated/prisma/client.js'

const db = new PrismaClient()

/** DocState'e yazar; documentName = Document.id. Yalnızca var olan
 *  dokümanlar için çağrılır (onAuthenticate doğrulamıştır). */
export class PrismaDatabase extends Database {
  async onLoadDocument({ documentName }: { documentName: string }): Promise<Buffer | null> {
    const state = await db.docState.findUnique({ where: { documentId: documentName } })
    return state ? state.ydoc : null
  }

  async onStoreDocument({ documentName, state }: { documentName: string; state: Uint8Array }): Promise<void> {
    const ydoc = Buffer.from(state)
    await db.docState.upsert({
      where: { documentId: documentName },
      create: { documentId: documentName, ydoc },
      update: { ydoc },
    })
  }
}
```

```ts
// collab/src/auth.ts
import { PrismaClient } from '../generated/prisma/client.js'

const db = new PrismaClient()

export type CollabActor = {
  id: string
  globalRole: 'ADMIN' | 'MEMBER'
  isActive: boolean
  memberships: { channelId: string; channelRole: 'LEAD' | 'MEMBER' }[]
}

/**
 * Auth.js DB oturumunda sessionToken == httpOnly çerez değeridir.
 * Provider, `token` seçeneğini websocket handshake'inde gönderir;
 * burada karşılığı Session tablosunda aranır.
 */
export async function resolveActor(token: string | null): Promise<CollabActor | null> {
  if (!token) return null
  const session = await db.session.findUnique({
    where: { sessionToken: token },
    include: { user: { include: { memberships: true } } },
  })
  if (!session || session.expires < new Date() || !session.user.isActive) return null
  return {
    id: session.user.id,
    globalRole: session.user.globalRole,
    isActive: session.user.isActive,
    memberships: session.user.memberships.map((m) => ({ channelId: m.channelId, channelRole: m.channelRole })),
  }
}
```

```ts
// collab/src/server.ts
import { Server } from '@hocuspocus/server'
import { Logger } from '@hocuspocus/extension-logger'
import { PrismaClient } from '../generated/prisma/client.js'
import { PrismaDatabase } from './database.js'
import { resolveActor } from './auth.js'
import { extractTextFromYDoc } from '../web-deps/doc-search-text.js'

const db = new PrismaClient()

const server = Server.configure({
  port: Number(process.env.PORT ?? 1234),
  extensions: [new Logger(), new PrismaDatabase()],

  async onAuthenticate({ token, documentName }) {
    const actor = await resolveActor(token)
    if (!actor) throw new Error('Oturum geçersiz')

    const doc = await db.document.findUnique({ where: { id: documentName } })
    if (!doc || doc.archivedAt) throw new Error('Doküman bulunamadı')
    const channel = await db.channel.findUniqueOrThrow({ where: { id: doc.channelId } })

    // Yalnızca YAZABİLENLER websocket açar. VIEW paylaşımı ayrı akıştır
    // (editör sayfası sunucu verisiyle salt-okur render eder).
    if (!can(actor, 'document:write', docResource(doc, channel))) {
      throw new Error('Bu dokümanı düzenleme yetkin yok')
    }
    return { actorId: actor.id }
  },

  async onStoreDocument({ documentName, doc }) {
    // Arama metni: her kayıtta güncelle. OnStore sıklaşırsa debounce'lanır;
    // v1'de Hocuspocus varsayılanı (bağlantı kopuşu + aralıklı) yeterlidir.
    const text = extractTextFromYDoc(doc)
    await db.document.update({
      where: { id: documentName },
```

`Action` birleşimine eklenir: `'document:create' | 'document:read' | 'document:write' | 'document:share' | 'document:delete'`.

`Resource` birleşimine eklenir:

```ts
| {
    kind: 'document'
    channelId: string
    channelVisibility: Visibility
    channelArchivedAt: Date | null
    documentVisibility: 'PUBLIC' | 'CHANNEL' | 'PRIVATE'
    ownerId?: string | null
    shared?: boolean
    sharedEdit?: boolean
  }
| { kind: 'document:new'; channelId: string; channelVisibility: Visibility }
```

`can()` içine (mevcut `default` ağından önce):

```ts
case 'document:create':
  if (resource.kind !== 'document:new') return false
  if (resource.channelArchivedAt) return false
  return isAdmin(actor) || !!membership(actor, resource.channelId)

case 'document:read': {
  if (resource.kind !== 'document') return false
  if (resource.channelArchivedAt) return false
  if (isAdmin(actor)) return true
  if (resource.documentVisibility === 'PUBLIC') return true
  if (resource.documentVisibility === 'CHANNEL') {
    return !!membership(actor, resource.channelId)
  }
  // PRIVATE: sahip + paylaşılan + admin
  return resource.ownerId === actor.id || !!resource.shared
}

case 'document:write': {
  if (resource.kind !== 'document') return false
  if (resource.channelArchivedAt) return false
  if (isAdmin(actor)) return true
  if (resource.ownerId === actor.id) return true
  if (!!resource.sharedEdit) return true
  // CHANNEL/OPEN üyeler kanal içinde yazar; PRIVATE dokümanda kanal üyeliği
  // yazma vermez (paylaşılan değilse görünmüyor zaten).
  if (resource.documentVisibility === 'PRIVATE') return false
  return !!membership(actor, resource.channelId)
}

case 'document:share':
  if (resource.kind !== 'document') return false
  if (isAdmin(actor)) return true
  if (resource.ownerId === actor.id) return true
  return membership(actor, resource.channelId)?.channelRole === 'LEAD'

case 'document:delete':
  if (resource.kind !== 'document') return false
  if (isAdmin(actor)) return true
  if (resource.ownerId === actor.id) return true
  return membership(actor, resource.channelId)?.channelRole === 'LEAD'
```

```ts
// src/server/documents.ts  ('use server')
import { z } from 'zod'
import { db } from '@/lib/db'
import { defineAction } from '@/lib/action'
import { getActor } from '@/lib/auth/session'
import { can, type Actor, type Visibility } from '@/lib/auth/policy'
import { recordActivity } from '@/lib/activity'
import { actionError } from '@/lib/action'

async function channelOf(actor: Actor, channelId: string) {
  const channel = await db.channel.findUnique({ where: { id: channelId } })
  if (!channel) throw actionError('NOT_FOUND')
  return channel
}

function docResource(doc: { channelId: string; visibility: string; ownerId: string }, channel: { visibility: Visibility; archivedAt: Date | null }) {
  return {
    kind: 'document' as const,
    channelId: doc.channelId,
    channelVisibility: channel.visibility,
    channelArchivedAt: channel.archivedAt,
    documentVisibility: doc.visibility as 'PUBLIC' | 'CHANNEL' | 'PRIVATE',
    ownerId: doc.ownerId,
  }
}

export const createDocument = defineAction({
  input: z.object({
    title: z.string().trim().max(120).default(''),
    channelId: z.string().cuid(),
    visibility: z.enum(['PUBLIC', 'CHANNEL', 'PRIVATE']).default('PUBLIC'),
    parentId: z.string().cuid().nullable().default(null),
  }),
  getActor,
  authorize: async ({ actor, input }) => {
    const channel = await channelOf(actor, input.channelId)
    return { allowed: can(actor, 'document:create', { kind: 'document:new', channelId: input.channelId, channelVisibility: channel.visibility }) }
  },
  handler: async ({ actor, input }) => {
    const doc = await db.document.create({
      data: {
        title: input.title,
        channelId: input.channelId,
        visibility: input.visibility,
        parentId: input.parentId,
        createdById: actor.id,
      },
    })
    await recordActivity(db, {
      actorId: actor.id, verb: 'document.created',
      entityType: 'Document', entityId: doc.id, channelId: input.channelId,
      meta: { title: doc.title },
    })
    return { id: doc.id }
  },
})

export const archiveDocument = defineAction({
  input: z.object({ id: z.string().cuid() }),
  getActor,
  authorize: async ({ actor, input }) => {
    const doc = await db.document.findUnique({ where: { id: input.id } })
    if (!doc) return { allowed: false, code: 'NOT_FOUND' as const }
    const channel = await channelOf(actor, doc.channelId)
    return { allowed: can(actor, 'document:delete', docResource(doc, channel)) }
  },
  handler: async ({ actor, input }) => {
    await db.document.update({ where: { id: input.id }, data: { archivedAt: new Date() } })
    await recordActivity(db, {
      actorId: actor.id, verb: 'document.archived',
      entityType: 'Document', entityId: input.id,
    })
    return { id: input.id }
  },
})

export const restoreDocument = defineAction({
  input: z.object({ id: z.string().cuid() }),
  getActor,
  authorize: async ({ actor }) => ({ allowed: can(actor, 'document:delete', { kind: 'member', id: '' }) || actor.globalRole === 'ADMIN' }),
  handler: async ({ actor, input }) => {
    // Çöp kutusu yalnızca admin ekranı: geri alma da admin yetkisi.
    await db.document.update({ where: { id: input.id }, data: { archivedAt: null } })
    await recordActivity(db, {
      actorId: actor.id, verb: 'document.restored',
      entityType: 'Document', entityId: input.id,
    })
    return { id: input.id }
  },
})

    title: z.string().trim().max(120).default(''),
    icon: z.string().trim().max(8).nullable().default(null),
    visibility: z.enum(['PUBLIC', 'CHANNEL', 'PRIVATE']).optional(),

  getActor,
  authorize: async ({ actor, input }) => {
```

```ts
// src/server/documents-query.ts  ('use server' YOK)
import { db } from '@/lib/db'
import { getActor } from '@/lib/auth/session'
import { type Actor } from '@/lib/auth/policy'

export async function listVisibleDocuments(): Promise<Awaited<ReturnType<typeof db.document.findMany>>> {
  const actor = await getActor()
  if (!actor) return []
  const mine = actor.memberships.map((m) => m.channelId)
  const docs = await db.document.findMany({
    where: { archivedAt: null, channel: { archivedAt: null } },
    include: { channel: { select: { visibility: true, name: true } } },
    orderBy: { updatedAt: 'desc' },
  })
  if (actor.globalRole === 'ADMIN') return docs
  return docs.filter((d) => {
    if (d.channel.visibility === 'OPEN') {
      if (d.visibility === 'PUBLIC') return true
      if (d.visibility === 'CHANNEL' && mine.includes(d.channelId)) return true
      // PRIVATE kontrolü paylaşım + sahiplik gerektirir; detay sayfası zaten can() uygular,
      // listede yalnızca okunabilenleri göster (detayda tekrar doğrulanır).
    }
    return mine.includes(d.channelId)
  })
}
```

```ts
// src/lib/rank.ts
// Lexorank: '0'..'z' karakterleriyle sözlük sıralı, aralık tükenince uzatma.
const CHARS = '0123456789abcdefghijklmnopqrstuvwxyz'
const MIN = '0'
const MAX = 'z'

export function rankBetween(prev: string, next: string): string {
  const p = prev === '' ? '' : prev
  const n = next === '' ? '' : next
  if (p === '' && n === '') return '0'
  if (n === '') return increment(p)  // sona ekle
  // ortak ön-ek atılır
  let i = 0
  while (i < p.length && i < n.length && p[i] === n[i]) i++
  const pRest = p.slice(i)
  const nRest = n.slice(i)
  // basit ara değer; karmaşıklık büyürse '0'ı tekrar ekleyerek uzatılır
  for (const mid of [CHARS[Math.floor(CHARS.length / 2)] ?? 'f', 'f0', 'f00', 'f000']) {
    const candidate = p.slice(0, i) + mid
    if (candidate > p && candidate < n) return candidate
  }
  return p + '0' // son çare: eşitse yanına ekle
}

export function increment(s: string): string {
  const chars = s.split('')
  let i = chars.length - 1
  while (i >= 0) {
    const idx = CHARS.indexOf(chars[i] ?? '')
    if (idx < CHARS.length - 1) {
      chars[i] = CHARS[idx + 1] ?? '0'
      return chars.join('')
    }
    chars[i] = MIN
    i--
  }
  return '1' + chars.join('')
}
```

```ts
// src/lib/storage.ts
import { mkdir, readFile, unlink, writeFile } from 'node:fs/promises'
import { randomUUID } from 'node:crypto'
import path from 'node:path'

const DIR = process.env.UPLOAD_DIR ?? path.join(process.cwd(), 'uploads')

export async function saveFile(file: { name: string; data: Buffer }): Promise<string> {
  await mkdir(DIR, { recursive: true })
  const storagePath = path.join(DIR, randomUUID())
  await writeFile(storagePath, file.data)
  return storagePath
}

export async function loadFile(storagePath: string): Promise<Buffer | null> {
  try {
    return await readFile(storagePath)
  } catch {
    return null
  }
}

export async function deleteFile(storagePath: string): Promise<void> {
  await unlink(storagePath).catch(() => {})
}
```
